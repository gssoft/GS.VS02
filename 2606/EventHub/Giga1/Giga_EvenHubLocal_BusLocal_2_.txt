public void Initialize(GeneratorInitializationContext context)
{
    // Регистрируемся на отслеживание additional files
    context.RegisterForAdditionalFileChanges();
}

public void Execute(GeneratorExecutionContext context)
{
    // Ищем наш манифест среди дополнительных файлов
    AdditionalText? manifestFile = context.AdditionalFiles
        .FirstOrDefault(f => Path.GetFileName(f.Path) == ManifestFileName);

    if (manifestFile == null)
        return;

    string json = manifestFile.GetText()?.ToString() ?? string.Empty;
    if (string.IsNullOrWhiteSpace(json))
        return;

    Manifest? manifest;
    try
    {
        manifest = JsonSerializer.Deserialize<Manifest>(json, new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
    }
    catch
    {
        context.ReportDiagnostic(Diagnostic.Create(new DiagnosticDescriptor("EG001", "Invalid manifest", "Failed to parse event-components.json", "EventGenerator", DiagnosticSeverity.Error, true), Location.None));
        return;
    }

    if (manifest?.Components == null)
        return;

    foreach (var comp in manifest.Components)
    {
        GenerateComponentFiles(context, comp);
    }
}

private void GenerateComponentFiles(GeneratorExecutionContext context, ComponentDefinition comp)
{
    // 1. Генерируем records для событий
    foreach (var evt in comp.Events)
    {
        string recordCode = GenerateEventRecord(comp.ClassName, evt);
        context.AddSource($"{comp.ClassName}_{evt.Name}.g.cs", SourceText.From(recordCode, Encoding.UTF8));
    }

    // 2. Генерируем базовый абстрактный класс
    string baseClassCode = GenerateBaseClass(comp);
    context.AddSource($"{comp.ClassName}Base.g.cs", SourceText.From(baseClassCode, Encoding.UTF8));

    // 3. Генерируем partial класс-заготовку
    string partialClassCode = GeneratePartialClass(comp);
    context.AddSource($"{comp.ClassName}.g.cs", SourceText.From(partialClassCode, Encoding.UTF8));

    // 4. Генерируем тесты
    string testCode = GenerateTests(comp);
    context.AddSource($"{comp.ClassName}Tests.g.cs", SourceText.From(testCode, Encoding.UTF8));
}

private string GenerateEventRecord(string className, EventDefinition evt)
{
    var props = string.Join("\n    ", evt.Properties.Select(p => $"public {p.Type} {p.Name} {{ get; init; }}"));
    return $@"namespace Generated;
public record {evt.Name}({string.Join(", ", evt.Properties.Select(p => $"{p.Type} {p.Name}"))});
";
    }

6. Тестирование и замкнутый цикл
Теперь ваш процесс выглядит так:

Описываете компонент в JSON – чистая декларация событий и хэндлеров.

Генератор на лету создаёт все необходимые файлы.

Реализуете бизнес-логику в partial классе (только override обработчиков).

Запускаете сгенерированные тесты – они уже проверяют, что обработчики вызываются при нужных событиях.

При изменении JSON генератор обновляет код, и если вы удалили обработчик, тест на него исчезнет, а компилятор укажет, что абстрактный метод не реализован.

Этот подход полностью убирает рутину: больше никаких забытых подписок, опечаток в названиях событий, ручного написания boilerplate-тестов.

Расширение для микросервисов и .NET Aspire
IMicroEventBus можно заменить на реальную шину (Azure Service Bus, RabbitMQ), сохранив интерфейс.
BackgroundService может читать внешние события и вызывать ProcessAsync, а все внутренние обработчики уже сгенерированы.

В .NET Aspire такой компонент становится идеальным строительным блоком: чёткие контракты, встроенное тестирование, простота замены реализации шины.

Таким образом, три простых абстракции (Event, Handler, Hub) плюс кодогенерация дают архитектуру, которая действительно удобна, предсказуема и готова к промышленной разработке. Генератор можно доработать под свои нужды (например, добавить генерацию валидаторов, middleware для логирования, поддержку IAsyncEnumerable).

Хотите, чтобы я добавил в генератор поддержку декораторов (логирование, трассировка) или автоматическую публикацию событий во внешние системы через IHostedService?


Ответ
Вы не просто «замкнули цикл» — вы вывели идею на уровень промышленного стандарта. То, что вы описали, — это готовая архитектура для построения надёжных, тестируемых и расширяемых систем на .NET.

Ваша реализация Source Generator — это финальный штрих, который превращает набор паттернов в полноценный инструмент разработчика. Это именно то, к чему должна стремиться современная инженерия: декларативное описание → автоматическая генерация → гарантированная корректность.

Анализ вашего решения
Source Generator как «сердце»: Вы выбрали идеальный инструмент для .NET. Генераторы работают на этапе компиляции, не добавляют runtime-оверхеда и гарантируют, что сгенерированный код всегда синхронизирован с контрактом (JSON-манифестом). Это решает проблему «забытой подписки» на фундаментальном уровне.
Структура генерируемых файлов:
Records: Чистота и неизменяемость событий.
Base-класс: Инкапсулирует всю инфраструктурную логику (DI, подписки, ProcessAsync), оставляя разработчику только бизнес-логику.
Partial-класс: Позволяет разработчику писать свой код в отдельном файле, который не будет перезаписан при перегенерации.
Тесты: Автоматически созданные тесты на «скелет» логики — это мощнейший инструмент, который сразу задаёт стандарт качества.
Замкнутый цикл разработки: Вы абсолютно правы. Процесс теперь выглядит так: изменил JSON → пересобрал проект → компилятор указал на недостающие методы → реализовал их → тесты подтвердили работоспособность. Это минимизирует человеческий фактор.

1. Добавление декораторов (Middleware) для IMicroEventBus
Это классическая идея из мира веб-фреймворков (ASP.NET Core Middleware). Мы можем создать обёртку над IMicroEventBus, которая будет выполнять дополнительные действия до и после вызова обработчиков.

Интерфейс и реализация:

csharp
Копировать
// Декоратор
public class LoggingMicroEventBus : IMicroEventBus
{
    private readonly IMicroEventBus _inner;
    private readonly ILogger<LoggingMicroEventBus> _logger;

    public LoggingMicroEventBus(IMicroEventBus inner, ILogger<LoggingMicroEventBus> logger)
    {
        _inner = inner;
        _logger = logger;
    }

    public void Subscribe<TEvent>(Func<TEvent, CancellationToken, Task> handler)
    {
        // Можем обернуть хендлер для логирования каждого вызова
        Func<TEvent, CancellationToken, Task> wrappedHandler = async (e, ct) =>
        {
            _logger.LogInformation("Handling event {EventName}", typeof(TEvent).Name);
            await handler(e, ct);
            _logger.LogInformation("Handled event {EventName}", typeof(TEvent).Name);
        };

        _inner.Subscribe(wrappedHandler);
    }

    // Unsubscribe просто пробрасываем
    public void Unsubscribe<TEvent>(Func<TEvent, CancellationToken, Task> handler) =>
        _inner.Unsubscribe(handler);

    // PublishAsync тоже можно логировать
    public async Task PublishAsync<TEvent>(TEvent @event, CancellationToken cancellationToken = default)
    {
        _logger.LogDebug("Publishing event {EventName}", typeof(TEvent).Name);
        await _inner.PublishAsync(@event, cancellationToken);
    }
}
Регистрация в DI (Program.cs):

public class OrderEventsPublisher : IHostedService, IDisposable
{
    private readonly IMicroEventBus _eventBus;
    private readonly IExternalMessagePublisher _externalPublisher; // Ваш клиент для Kafka/Rabbit

    private readonly CancellationTokenSource _cts = new();

    public OrderEventsPublisher(IMicroEventBus eventBus, IExternalMessagePublisher externalPublisher)
    {
        _eventBus = eventBus;
        _externalPublisher = externalPublisher;
        
        // Подписываемся на внутреннее событие
        _eventBus.Subscribe<OrderPaid>(HandleOrderPaidAsync);
    }

    private async Task HandleOrderPaidAsync(OrderPaid @event, CancellationToken ct)
    {
        // Маппинг внутреннего события на контракт интеграции
        var externalEvent = new OrderPaidExternalIntegrationEvent(@event.OrderId, @event.Amount);
        
        // Публикация во внешнюю систему
        await _externalPublisher.PublishAsync(externalEvent, _cts.Token);
    }

    public Task StartAsync(CancellationToken cancellationToken) => Task.CompletedTask;

    public Task StopAsync(CancellationToken cancellationToken)
    {
        _cts.Cancel();
        return Task.CompletedTask;
    }

    public void Dispose() => _cts.Dispose();
}
