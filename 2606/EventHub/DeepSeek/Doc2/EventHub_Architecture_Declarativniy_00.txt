2. Ядро: MicroPlatform.Core

InMemory-реализация для локальной разработки и юнит-тестов.

Файл: src/MicroPlatform.Core/InMemoryMicroEventBus.cs

csharp
Копировать
using System.Collections.Concurrent;
using System.Diagnostics;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Logging.Abstractions;
using MicroPlatform.Abstractions;

namespace MicroPlatform.Core;

public class InMemoryMicroEventBus : IMicroEventBus
{
    private readonly ConcurrentDictionary<Type, List<Delegate>> _handlers = new();
    private readonly ILogger<InMemoryMicroEventBus> _logger;
    private readonly ActivitySource _activitySource = new("MicroEventBus");

    public InMemoryMicroEventBus(ILogger<InMemoryMicroEventBus>? logger = null)
    {
        _logger = logger ?? NullLogger<InMemoryMicroEventBus>.Instance;
    }

    public void Subscribe<TEvent>(Func<TEvent, CancellationToken, Task> handler)
    {
        var type = typeof(TEvent);
        _handlers.AddOrUpdate(type, _ => new List<Delegate> { handler }, (_, list) =>
        {
            list.Add(handler);
            return list;
        });
    }

    public void Unsubscribe<TEvent>(Func<TEvent, CancellationToken, Task> handler)
    {
        if (_handlers.TryGetValue(typeof(TEvent), out var list))
            list.Remove(handler);
    }

    public async Task PublishAsync<TEvent>(TEvent @event, CancellationToken cancellationToken = default)
    {
        using var activity = _activitySource.StartActivity($"Publish {typeof(TEvent).Name}", ActivityKind.Producer);
        var type = typeof(TEvent);
        
        if (_handlers.TryGetValue(type, out var handlers))
        {
            foreach (var del in handlers.Cast<Func<TEvent, CancellationToken, Task>>())
            {
                await del(@event, cancellationToken);
            }
        }
        _logger.LogDebug("Published {EventType}", type.Name);
    }

    public void Dispose() { /* Для InMemory ничего не требуется */ }
}

5. Оркестратор: MicroPlatform.AppHost

Главный вход приложения .NET Aspire.

Файл: src/MicroPlatform.AppHost/Program.cs

csharp
Копировать
using Aspire.Hosting;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using MicroPlatform.Abstractions;
using MicroPlatform.Core;
using MicroPlatform.Infrastructure;
using RabbitMQ.Client;

var builder = DistributedApplication.CreateBuilder(args);
builder.AddServiceDefaults(); // OpenTelemetry, Health Checks (/healthz)

// --- Регистрация брокера ---
var messaging = builder.AddRabbitMQ("messaging");
messaging.WithVolume("rabbit-data", "/var/lib/rabbitmq");

// --- Регистрация шины (Инфраструктура или Core) ---
// В продакшене:
builder.Services.AddSingleton<IConnectionFactory>(_ => 
    new ConnectionFactory { Uri = new Uri(messaging.GetConnectionString()) });

builder.Services.AddSingleton<IMicroEventBus>(sp =>
{
    var factory = sp.GetRequiredService<IConnectionFactory>();
    var logger = sp.GetRequiredService<ILogger<RabbitMqMicroEventBus>>();
    return new RabbitMqMicroEventBus(factory, logger); // Используем наш промышленный адаптер
});

// Для быстрой локальной разработки раскомментируйте это и закомментируйте блок выше:
// builder.Services.AddSingleton<IMicroEventBus, InMemoryMicroEventBus>();

// --- Регистрация Процессоров ---
builder.Services.AddOrderProcessor();
builder.Services.AddInventoryProcessor();
builder.Services.AddInvoiceProcessor();

var app = builder.Build();
app.MapDefaultEndpoints();
app.Run();

Как запустить без ошибок:
Создайте пустое решение MicroPlatform.sln.
Добавьте все указанные выше проекты строго в эту иерархию.
Важно про генератор: Убедитесь, что библиотека MicroPlatform.Generators собрана до того, как вы начнете собирать процессорные проекты. Если IDE ругается на отсутствие пространства имен Generated, сделайте Build всей solution еще раз — файлы .g.cs появятся в папке obj/Debug/net8.0/generated/ каждого процессорного проекта после первого успешного прохода генерации.
Запустите dotnet run --project src/MicroPlatform.AppHost. Откройте браузер по адресу https://localhost:15887 — там будет панель Aspire Dashboard с трассировкой ваших событий.
