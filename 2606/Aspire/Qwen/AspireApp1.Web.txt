using AspireApp1.Web;
using AspireApp1.Web.Components;

var builder = WebApplication.CreateBuilder(args);

// Add service defaults & Aspire client integrations.
builder.AddServiceDefaults();
builder.AddRedisOutputCache("cache");

// Add services to the container.
builder.Services.AddRazorComponents()
    .AddInteractiveServerComponents();

builder.Services.AddHttpClient<WeatherApiClient>(client =>
    {
        // This URL uses "https+http://" to indicate HTTPS is preferred over HTTP.
        // Learn more about service discovery scheme resolution at https://aka.ms/dotnet/sdschemes.
        client.BaseAddress = new("https+http://apiservice");
    });

var app = builder.Build();

if (!app.Environment.IsDevelopment())
{
    app.UseExceptionHandler("/Error", createScopeForErrors: true);
    // The default HSTS value is 30 days. You may want to change this for production scenarios, see https://aka.ms/aspnetcore-hsts.
    app.UseHsts();
}

app.UseHttpsRedirection();

app.UseAntiforgery();

app.UseOutputCache();

app.MapStaticAssets();

app.MapRazorComponents<App>()
    .AddInteractiveServerRenderMode();

app.MapDefaultEndpoints();

app.Run();

namespace AspireApp1.Web;

public class WeatherApiClient(HttpClient httpClient)
{
    public async Task<WeatherForecast[]> GetWeatherAsync(int maxItems = 10, CancellationToken cancellationToken = default)
    {
        List<WeatherForecast>? forecasts = null;

        await foreach (var forecast in httpClient.GetFromJsonAsAsyncEnumerable<WeatherForecast>("/weatherforecast", cancellationToken))
        {
            if (forecasts?.Count >= maxItems)
            {
                break;
            }
            if (forecast is not null)
            {
                forecasts ??= [];
                forecasts.Add(forecast);
            }
        }

        return forecasts?.ToArray() ?? [];
    }
}

public record WeatherForecast(DateOnly Date, int TemperatureC, string? Summary)
{
    public int TemperatureF => 32 + (int)(TemperatureC / 0.5556);
}

    private WeatherForecast[]? _forecasts;
    private bool _isLoading = true;
    private bool _isRefreshing;
    private string? _error;
    private DateTime _lastUpdate;

    // Настройки автообновления
    private bool _autoRefresh = true;
    private int _intervalSeconds = 5;

    // Управление жизненным циклом таймера
    private PeriodicTimer? _timer;
    private CancellationTokenSource? _cts;

    protected override async Task OnInitializedAsync()
    {
        // Первичная загрузка
        await LoadWeatherAsync(isInitial: true);

        // Запускаем цикл автообновления
        StartAutoRefresh();
    }

    private void StartAutoRefresh()
    {
        // Останавливаем предыдущий таймер, если был
        StopAutoRefresh();

        _cts = new CancellationTokenSource();
        _timer = new PeriodicTimer(TimeSpan.FromSeconds(_intervalSeconds));
        var token = _cts.Token;

        // Запускаем фоновый цикл (fire-and-forget)
        _ = Task.Run(async () =>
        {
            try
            {
                while (await _timer.WaitForNextTickAsync(token))
                {
                    if (!_autoRefresh) continue;

                    await LoadWeatherAsync(isInitial: false);

                    // ВАЖНО: обновляем UI через InvokeAsync,
                    // т.к. мы находимся не в UI-потоке Blazor
                    await InvokeAsync(StateHasChanged);
                }
            }
            catch (OperationCanceledException)
            {
                // Нормальное завершение при остановке — ничего не делаем
            }
            catch (Exception ex)
            {
                // Логируем, но не даём циклу упасть
                Console.WriteLine($"[AutoRefresh] Ошибка: {ex.Message}");
            }
        }, token);
    }

    private void StopAutoRefresh()
    {
        _cts?.Cancel();
        _timer?.Dispose();
        _cts?.Dispose();
        _timer = null;
        _cts = null;
    }

    private async Task LoadWeatherAsync(bool isInitial)
    {
        if (isInitial)
        {
            _isLoading = true;
        }
        else
        {
            _isRefreshing = true;
        }

        _error = null;

        try
        {
            _forecasts = await WeatherApiClient.GetWeatherAsync(maxItems: 10);
            _lastUpdate = DateTime.Now;
        }
        catch (Exception ex)
        {
            _error = ex.Message;
        }
        finally
        {
            _isLoading = false;
            _isRefreshing = false;
        }
    }

    private async Task ManualRefreshAsync()
    {
        await LoadWeatherAsync(isInitial: false);
    }

    private void OnAutoRefreshChanged()
    {
        if (_autoRefresh && _timer is null)
        {
            StartAutoRefresh();
        }
    }

    private void OnIntervalChanged()
    {
        // При смене интервала — перезапускаем таймер
        if (_autoRefresh)
        {
            StartAutoRefresh();
        }
    }

    private string GetBadgeColor(int tempC) => tempC switch
    {
        < 0 => "bg-primary",
        < 10 => "bg-info",
        < 20 => "bg-success",
        < 30 => "bg-warning text-dark",
        _ => "bg-danger"
    };

    // ВАЖНО: освобождаем ресурсы при уходе со страницы
    public ValueTask DisposeAsync()
    {
        StopAutoRefresh();
        return ValueTask.CompletedTask;
    }
}
