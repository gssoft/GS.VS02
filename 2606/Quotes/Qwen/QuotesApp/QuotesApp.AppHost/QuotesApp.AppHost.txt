    .WithReference(redis)
    .WaitFor(redis);

var apiService = builder.AddProject<Projects.QuotesApp_ApiService>("apiservice")
    .WithReference(redis)
    .WaitFor(redis);

    .WithReference(apiService)
    .WithReference(redis)
    .WaitFor(apiService)
    .WithExternalHttpEndpoints();

builder.Build().Run();
